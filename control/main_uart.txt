#include "uart_manager.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include <stdio.h>

static int16_t inputs[4];   // desde MATLAB → ESP32
static int16_t outputs[4];  // desde ESP32 → MATLAB

extern "C" void app_main(void)
{
    start_uart_task(inputs, outputs);

    while (true)
    {
        // Simula tus sensores o estados
        outputs[0] = inputs[0];
        outputs[1] = inputs[1];
        outputs[2] = inputs[2];
        outputs[3] = inputs[3];

        vTaskDelay(pdMS_TO_TICKS(20)); // 10 Hz
    }
}
