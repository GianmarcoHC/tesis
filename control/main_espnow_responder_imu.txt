#include <stdio.h>
#include <string.h>
#include "freertos/FreeRTOS.h"
#include "freertos/Task.h"
#include "esp_now.h"
#include "esp_wifi.h"
#include "esp_netif.h"
#include "esp_mac.h"
#include "esp_event.h"
#include "nvs_flash.h"
#include "esp_log.h"

#define ESP_CHANNEL 1

typedef struct __attribute__((packed)) {
    int16_t roll_x100;   // roll * 100
    int16_t pitch_x100;  // pitch * 100
} imu_packet_t;

// Recibir desde cualquiera
static uint8_t peer_mac[ESP_NOW_ETH_ALEN] = {0xFF,0xFF,0xFF,0xFF,0xFF,0xFF};

static const char *TAG = "esp_now_resp";

static esp_err_t init_wifi(void)
{
    wifi_init_config_t wifi_init_config = WIFI_INIT_CONFIG_DEFAULT();

    ESP_ERROR_CHECK(esp_netif_init());
    ESP_ERROR_CHECK(esp_event_loop_create_default());
    ESP_ERROR_CHECK(nvs_flash_init());
    ESP_ERROR_CHECK(esp_wifi_init(&wifi_init_config));
    ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_STA));
    ESP_ERROR_CHECK(esp_wifi_start());

    ESP_LOGI(TAG, "WiFi init OK");
    return ESP_OK;
}

void recv_cb(const esp_now_recv_info_t *info, const uint8_t *data, int data_len)
{
    if (data_len == sizeof(imu_packet_t)) {

        const imu_packet_t *pkt = (const imu_packet_t *)data;

        float roll  = pkt->roll_x100  / 100.0f;
        float pitch = pkt->pitch_x100 / 100.0f;

        ESP_LOGI(TAG,
                 "From " MACSTR " → roll=%.2f°, pitch=%.2f°",
                 MAC2STR(info->src_addr),
                 roll, pitch);

    } else {
        ESP_LOGW(TAG, "Unexpected data size: %d", data_len);
    }
}

void send_cb(const esp_now_send_info_t *info, esp_now_send_status_t status)
{
    if (status == ESP_NOW_SEND_SUCCESS)
        ESP_LOGI(TAG, "Send OK");
    else
        ESP_LOGW(TAG, "Send FAIL");
}

static esp_err_t init_esp_now(void)
{
    ESP_ERROR_CHECK(esp_now_init());
    esp_now_register_recv_cb(recv_cb);
    esp_now_register_send_cb(send_cb);

    ESP_LOGI(TAG, "ESP-NOW init OK");
    return ESP_OK;
}

static esp_err_t register_peer(uint8_t *peer_addr)
{
    esp_now_peer_info_t peer = {};
    memcpy(peer.peer_addr, peer_addr, ESP_NOW_ETH_ALEN);
    peer.channel = ESP_CHANNEL;
    peer.ifidx   = WIFI_IF_STA;

    return esp_now_add_peer(&peer);
}

extern "C" void app_main(void)
{
    ESP_ERROR_CHECK(init_wifi());
    ESP_ERROR_CHECK(init_esp_now());

    // Recibir de cualquiera → no registrar peer
    // Si quisieras responder a un solo sender, usa:
    // ESP_ERROR_CHECK(register_peer(peer_mac));
}
