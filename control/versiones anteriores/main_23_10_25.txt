#include <stdio.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "driver/ledc.h"
#include "driver/gpio.h"
#include "esp_err.h"

#define ENCODER_A_GPIO   32   //
#define ENCODER_B_GPIO   33

#define CPR_ENCODER      13       // counts per revolution del encoder
#define GEAR_RATIO       75
#define COUNTS_PER_REV   (CPR_ENCODER * GEAR_RATIO * 2)  // 4x decodificación

#define PWM_GPIO   25   // Pin PWM -> M1PWM
#define DIR_GPIO   26   // Pin dirección -> M1DIR

#define PWM_FREQ_HZ     20000          // 20 kHz
#define PWM_RES_BITS    LEDC_TIMER_10_BIT
#define PWM_MAX_DUTY    ((1 << PWM_RES_BITS) - 1)
#define PWM_CHANNEL     LEDC_CHANNEL_0
#define PWM_TIMER       LEDC_TIMER_0
#define PWM_MODE        LEDC_LOW_SPEED_MODE

static volatile int32_t encoder_count = 0;

// ===============================
// Variables globales del PID
// ===============================
float Kp = 9.9518f;
float Ki = 0.0031f;
float Kd = 0.4645f;

//float Kp = 1.0f;
//float Ki = 0.0f;
//float Kd = 0.0f;

float integral = 0.0f;
float prev_error = 0.0f;

float setpoint = 180.0f;   // referencia en rev/s
float Ts = 0.02f;        // periodo de muestreo (20 ms)

// ===============================
// ISR de canal A
// ===============================
static void IRAM_ATTR encoder_isr(void* arg) {
    int a = gpio_get_level(ENCODER_A_GPIO);
    int b = gpio_get_level(ENCODER_B_GPIO);

    // Decodificación cuadratura simple
    if (a == b) {
        encoder_count++;   // una dirección
    } else {
        encoder_count--;   // dirección contraria
    }
}

// ===============================
// Inicializa GPIOs e interrupciones
// ===============================
static void encoder_init(void) {
    gpio_config_t io_conf = {
        .pin_bit_mask = (1ULL << ENCODER_A_GPIO) | (1ULL << ENCODER_B_GPIO),
        .mode = GPIO_MODE_INPUT,
        .pull_up_en = GPIO_PULLUP_ENABLE,
        .pull_down_en = 0,
        .intr_type = GPIO_INTR_DISABLE
    };
    gpio_config(&io_conf);

    // Configura interrupción solo en canal A
    gpio_set_intr_type(ENCODER_A_GPIO, GPIO_INTR_ANYEDGE);
    gpio_install_isr_service(0);
    gpio_isr_handler_add(ENCODER_A_GPIO, encoder_isr, NULL);
}

// Inicializa PWM
static void pwm_init(void) {
    ledc_timer_config_t timer_conf = {
        .speed_mode = PWM_MODE,
        .timer_num = PWM_TIMER,
        .duty_resolution = PWM_RES_BITS,
        .freq_hz = PWM_FREQ_HZ,
        .clk_cfg = LEDC_AUTO_CLK
    };
    ledc_timer_config(&timer_conf);

    ledc_channel_config_t ch_conf = {
        .gpio_num = PWM_GPIO,
        .speed_mode = PWM_MODE,
        .channel = PWM_CHANNEL,
        .timer_sel = PWM_TIMER,
        .duty = 0,
        .hpoint = 0
    };
    ledc_channel_config(&ch_conf);
}

// Ajusta duty (0.0 a 1.0)
static void pwm_set(float duty) {
    if (duty < 0) duty = 0;
    if (duty > 1) duty = 1;
    uint32_t duty_val = (uint32_t)(duty * PWM_MAX_DUTY);
    ledc_set_duty(PWM_MODE, PWM_CHANNEL, duty_val);
    ledc_update_duty(PWM_MODE, PWM_CHANNEL);
}

// Control con valor entre -1.0 y 1.0
static void motor_set(float pwm_val) {
    if (pwm_val > 0.0f) {
        // forward (antihorario en tu caso)
        gpio_set_level(DIR_GPIO, 0);      // 0 = forward
        pwm_set(pwm_val);
    } else if (pwm_val < 0.0f) {
        // reverse (horario)
        gpio_set_level(DIR_GPIO, 1);      // 1 = reverse
        pwm_set(1.0f + pwm_val);                // usa valor absoluto
    } else {
        pwm_set(0.0f);
    }
}

void app_main(void) {
    // Configura pin de dirección
    gpio_config_t dir_conf = {
        .pin_bit_mask = (1ULL << DIR_GPIO),
        .mode = GPIO_MODE_OUTPUT,
        .pull_up_en = 0,
        .pull_down_en = 0,
        .intr_type = GPIO_INTR_DISABLE
    };
    gpio_config(&dir_conf);

    pwm_init();
    encoder_init();

    while (1) {
        // ======== 1. Calcular posición a partir del encoder ========
        int32_t count = encoder_count;
        float revolutions = (float)count / COUNTS_PER_REV;
        float angle_deg = revolutions * 360.0f;

        // ======== 2. Calcular PID ========
        float error = setpoint - angle_deg;

        // Integral con anti-windup
        integral += error * Ts;
        //if (integral > 5.0f) integral = 5.0f;
        //if (integral < -5.0f) integral = -5.0f;

        float derivative = (error - prev_error) / Ts;
        prev_error = error;

        float output = (Kp * error + Ki * integral + Kd * derivative)/100;

        // Limitar a rango [-1, 1]
        if (output > 0.3f) output = 0.3f;
        if (output < -0.3f) output = -0.3f;

        // ======== 3. Aplicar al motor ========
        motor_set(output);

        // ======== 4. Mostrar datos ========
        printf("Ref: %.2f\tPos: %.2f\tPWM: %.2f\n", setpoint, angle_deg, output);

        vTaskDelay(pdMS_TO_TICKS(20));  // periodo de muestreo
    }
}
